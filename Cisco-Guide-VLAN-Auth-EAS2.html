access-list rules_LAN remark SSH sur srv-web et srv-web2
access-list rules_LAN permit tcp 192.168.1.0 255.255.255.0 host 192.168.2.254 eq 22
access-list rules_LAN permit tcp 192.168.1.0 255.255.255.0 host 192.168.2.253 eq 22

access-list rules_LAN remark Ping ok vers 1.1.1.1
access-list rules_LAN permit icmp 192.168.1.0 255.255.255.0 host 1.1.1.1 echo

access-list rules_LAN remark Resolution DNS
access-list rules_LAN permit udp 192.168.1.0 255.255.255.0 host 1.1.1.1 eq 53

access-list rules_LAN remark Navigation web
access-list rules_LAN permit tcp 192.168.1.0 255.255.255.0 any eq 80
access-list rules_LAN permit tcp 192.168.1.0 255.255.255.0 any eq 443

access-list rules_LAN remark Synchronisation horloge
access-list rules_LAN permit udp 192.168.1.0 255.255.255.0 host 51.145.123.29 eq 123

access-list rules_LAN remark SSH vers ASA
access-list rules_LAN permit tcp 192.168.1.0 255.255.255.0 interface inside eq 22
access-list rules_LAN permit tcp 192.168.1.0 255.255.255.0 host 192.168.2.254 eq 22
 
access-group rules_LAN in interface inside
